cmake_minimum_required(VERSION 3.28)
project(prox_mpc_msgs)

find_package(ament_cmake REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)

# Compiler warnings, matching upstream Nav2's package-level policy
# (nav2_common/cmake/nav2_package.cmake), applied directory-wide exactly as
# nav2_msgs does it (before rosidl_generate_interfaces() below, so the
# generated typesupport targets are held to the same bar as hand-written code).
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU" OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Werror -Wdeprecated -fPIC -Wshadow -Wnull-dereference)
  # GCC 13 raises -Wnull-dereference inside libstdc++'s own <regex> headers
  # (stl_algobase.h, reached through regex_traits::transform_primary) once
  # --coverage changes its inlining decisions, which is how the coverage build
  # hits it and an ordinary build does not. The report is attributed to the
  # translation unit that instantiates the template, so -Wno-system-headers
  # does not reach it. Keep the diagnostic, but do not let a false positive in
  # the standard library fail the build.
  add_compile_options(-Wno-error=null-dereference)
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-Wnon-virtual-dtor>")
endif()

rosidl_generate_interfaces(${PROJECT_NAME}
  msg/Obstacle.msg
  msg/ObstacleArray.msg
  msg/SolverDiagnostics.msg
  DEPENDENCIES geometry_msgs std_msgs
)

ament_export_dependencies(rosidl_default_runtime)

if(BUILD_TESTING)
  find_package(ament_lint_auto REQUIRED)
  # uncrustify is the enforced C++ formatter across this repository; cpplint is
  # intentionally disabled everywhere else to avoid duplicating/contradicting
  # it (every other package's CMakeLists carries this same override). This
  # package never needed it until its first C++ test file.
  set(ament_cmake_cpplint_FOUND TRUE)
  # Copyright is disabled: interface-only package, no per-file headers;
  # full text in LICENSE.
  set(ament_cmake_copyright_FOUND TRUE)
  ament_lint_auto_find_test_dependencies()

  find_package(ament_cmake_gtest REQUIRED)
  find_package(rosidl_typesupport_introspection_cpp REQUIRED)

  # Wire-format pin: field names, order, types, defaults and the STATUS_*
  # constant values, generated by the rosidl_generate_interfaces() call above.
  # The introspection typesupport target is what defines the declaration-order
  # member list get_message_type_support_handle<T>() returns.
  rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
  rosidl_get_typesupport_target(
    introspection_typesupport_target ${PROJECT_NAME} rosidl_typesupport_introspection_cpp)
  ament_add_gtest(test_solver_diagnostics_wire test/test_solver_diagnostics_wire.cpp)
  target_link_libraries(test_solver_diagnostics_wire
    ${cpp_typesupport_target}
    ${introspection_typesupport_target}
  )
  ament_target_dependencies(test_solver_diagnostics_wire rosidl_typesupport_introspection_cpp)
endif()

ament_package()
